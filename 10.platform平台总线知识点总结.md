# Linux Platform平台总线驱动开发总结

## 第一部分：Platform平台总线开发步骤详解

### 1.1 平台总线概述

Platform平台总线是Linux内核提供的一种虚拟总线，用于管理那些不具备热插拔能力、无法自动枚举的外设（如SoC内部集成的控制器、GPIO、I2C、SPI等）。

**核心思想**：将设备信息（platform_device）和设备驱动（platform_driver）分离，通过总线进行匹配绑定。

**匹配机制优先级**（从高到低）：
1. **设备树匹配**：通过 `of_match_table` 中的 `compatible` 属性匹配
2. **ACPI匹配**：通过 `acpi_match_table` 匹配（x86平台常用）
3. **id_table匹配**：通过 `id_table` 中的 `name` 字段匹配
4. **name匹配**：通过驱动的 `driver.name` 与设备的 `name` 匹配

### 1.2 Platform Device（平台设备）开发步骤

#### 方式一：传统方式（C代码注册）

##### 步骤1：定义平台设备资源
```c
static struct resource xxx_resources[] = {
    [0] = {
        .start = 0x12345000,           /* 资源起始地址 */
        .end   = 0x12345FFF,           /* 资源结束地址 */
        .flags = IORESOURCE_MEM,       /* 资源类型：内存 */
    },
    [1] = {
        .start = 100,                  /* 中断号 */
        .end   = 100,
        .flags = IORESOURCE_IRQ,       /* 资源类型：中断 */
    },
};
```

##### 步骤2：定义平台设备结构体
```c
static struct platform_device xxx_device = {
    .name = "xxx-device",              /* 设备名称，用于匹配 */
    .id   = -1,                        /* 设备ID，-1表示只有一个设备 */
    .num_resources = ARRAY_SIZE(xxx_resources),
    .resource = xxx_resources,
    .dev = {
        .platform_data = &xxx_pdata,   /* 平台私有数据 */
        .release = xxx_release,        /* 设备释放函数, 这个函数必须实现*/
    },
};
```

##### 步骤3：注册平台设备
```c
static int __init xxx_device_init(void)
{
    return platform_device_register(&xxx_device);
}

static void __exit xxx_device_exit(void)
{
    platform_device_unregister(&xxx_device);
}

module_init(xxx_device_init);
module_exit(xxx_device_exit);
```

#### 方式二：设备树方式（推荐）

在设备树（.dts）文件中添加设备节点：
```dts
xxx_device: xxx@12345000 {
    compatible = "vendor,xxx-device";  /* 用于匹配驱动 */
    reg = <0x12345000 0x1000>;         /* 寄存器地址和大小 */
    interrupts = <GIC_SPI 100 IRQ_TYPE_LEVEL_HIGH>;
    clocks = <&cru CLK_XXX>;
    clock-names = "xxx_clk";
    status = "okay";
};
```

### 1.3 Platform Driver（平台驱动）开发步骤

#### 步骤1：定义设备树匹配表（推荐）
```c
static const struct of_device_id xxx_of_match[] = {
    { .compatible = "vendor,xxx-device", .data = NULL },
    { .compatible = "vendor,xxx-device-v2", .data = &xxx_v2_data },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(of, xxx_of_match);
```

#### 步骤2：定义ID匹配表（可选，传统方式）
```c
static const struct platform_device_id xxx_id_table[] = {
    { "xxx-device", 0 },
    { "xxx-device-v2", 1 },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(platform, xxx_id_table);
```

#### 步骤3：实现probe函数（核心）
```c
static int xxx_probe(struct platform_device *pdev)
{
    struct resource *res;
    void __iomem *base;
    int irq, ret;
    struct xxx_dev *dev;

    /* 1. 分配设备私有数据结构 */
    dev = devm_kzalloc(&pdev->dev, sizeof(*dev), GFP_KERNEL);
    if (!dev)
        return -ENOMEM;

    /* 2. 获取内存资源 */
    res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    base = devm_ioremap_resource(&pdev->dev, res);
    if (IS_ERR(base))
        return PTR_ERR(base);
    dev->base = base;

    /* 3. 获取中断资源 */
    irq = platform_get_irq(pdev, 0);
    if (irq < 0)
        return irq;
    
    ret = devm_request_irq(&pdev->dev, irq, xxx_irq_handler,
                           0, "xxx-irq", dev);
    if (ret)
        return ret;

    /* 4. 获取时钟资源 */
    dev->clk = devm_clk_get(&pdev->dev, "xxx_clk");
    if (IS_ERR(dev->clk))
        return PTR_ERR(dev->clk);
    clk_prepare_enable(dev->clk);

    /* 5. 获取GPIO资源 */
    dev->gpio = devm_gpiod_get(&pdev->dev, "reset", GPIOD_OUT_LOW);
    if (IS_ERR(dev->gpio))
        return PTR_ERR(dev->gpio);

    /* 6. 读取设备树属性 */
    of_property_read_u32(pdev->dev.of_node, "xxx-value", &dev->value);

    /* 7. 注册字符设备或其他子系统 */
    ret = xxx_register_chrdev(dev);
    if (ret)
        return ret;

    /* 8. 保存私有数据到platform_device */
    platform_set_drvdata(pdev, dev);

    dev_info(&pdev->dev, "xxx device probed successfully\n");
    return 0;
}
```

#### 步骤4：实现remove函数
```c
static int xxx_remove(struct platform_device *pdev)
{
    struct xxx_dev *dev = platform_get_drvdata(pdev);

    /* 注销字符设备 */
    xxx_unregister_chrdev(dev);

    /* 关闭时钟（如果未使用devm_） */
    clk_disable_unprepare(dev->clk);

    dev_info(&pdev->dev, "xxx device removed\n");
    return 0;
}
```

#### 步骤5：定义platform_driver结构体
```c
static struct platform_driver xxx_driver = {
    .probe  = xxx_probe,
    .remove = xxx_remove,
    .driver = {
        .name = "xxx-device",
        .of_match_table = xxx_of_match,
        .pm = &xxx_pm_ops,             /* 电源管理（可选） */
    },
    .id_table = xxx_id_table,          /* ID匹配表（可选） */
};
```

#### 步骤6：注册/注销平台驱动

**方式一：标准方式**
```c
static int __init xxx_driver_init(void)
{
    return platform_driver_register(&xxx_driver);
}

static void __exit xxx_driver_exit(void)
{
    platform_driver_unregister(&xxx_driver);
}

module_init(xxx_driver_init);
module_exit(xxx_driver_exit);
```

**方式二：简化宏（推荐）**
```c
module_platform_driver(xxx_driver);
```

### 1.4 电源管理（可选）

```c
static int xxx_suspend(struct device *dev)
{
    struct xxx_dev *xdev = dev_get_drvdata(dev);
    /* 保存寄存器状态，关闭时钟等 */
    clk_disable_unprepare(xdev->clk);
    return 0;
}

static int xxx_resume(struct device *dev)
{
    struct xxx_dev *xdev = dev_get_drvdata(dev);
    /* 恢复寄存器状态，使能时钟等 */
    clk_prepare_enable(xdev->clk);
    return 0;
}

static SIMPLE_DEV_PM_OPS(xxx_pm_ops, xxx_suspend, xxx_resume);
```

## 第二部分：常用结构体、函数和API

### 2.1 核心结构体

#### `struct platform_device` - 平台设备结构体

```c
struct platform_device {
    const char      *name;          /* 设备名称 */
    int             id;             /* 设备ID，-1表示唯一 */
    bool            id_auto;        /* 是否自动分配ID */
    struct device   dev;            /* 内嵌的device结构体 */
    u64             platform_dma_mask;
    struct device_dma_parameters dma_parms;
    u32             num_resources;  /* 资源数量 */
    struct resource *resource;      /* 资源数组指针 */
    const struct platform_device_id *id_entry; /* 匹配的ID表项 */
    char            *driver_override; /* 强制使用的驱动名称 */
    /* 以下为MFD相关 */
    struct mfd_cell *mfd_cell;
    struct pdev_archdata archdata;
};
```

**重要成员说明**：
| 成员 | 类型 | 说明 |
|------|------|------|
| name | const char * | 设备名称，用于与驱动匹配 |
| id | int | 设备实例ID，-1表示只有一个实例 |
| dev | struct device | 内嵌设备结构体，包含of_node等 |
| num_resources | u32 | 资源数量 |
| resource | struct resource * | 资源数组 |

---

#### `struct platform_driver` - 平台驱动结构体

```c
struct platform_driver {
    int (*probe)(struct platform_device *);     /* 设备匹配成功时调用 */
    int (*remove)(struct platform_device *);    /* 设备移除时调用 */
    void (*shutdown)(struct platform_device *); /* 系统关机时调用 */
    int (*suspend)(struct platform_device *, pm_message_t state); /* 休眠 */
    int (*resume)(struct platform_device *);    /* 唤醒 */
    struct device_driver driver;                /* 内嵌的device_driver */
    const struct platform_device_id *id_table;  /* ID匹配表 */
    bool prevent_deferred_probe;                /* 防止延迟探测 */
};
```

**重要成员说明**：
| 成员 | 类型 | 说明 |
|------|------|------|
| probe | 函数指针 | 匹配成功后的初始化函数 |
| remove | 函数指针 | 设备移除时的清理函数 |
| driver | struct device_driver | 包含name、of_match_table等 |
| id_table | struct platform_device_id * | 传统ID匹配表 |

---

#### `struct resource` - 资源描述结构体

```c
struct resource {
    resource_size_t start;  /* 资源起始地址/编号 */
    resource_size_t end;    /* 资源结束地址/编号 */
    const char *name;       /* 资源名称 */
    unsigned long flags;    /* 资源类型标志 */
    unsigned long desc;     /* 资源描述 */
    struct resource *parent, *sibling, *child;
};
```

**资源类型标志（flags）**：
| 标志 | 值 | 说明 |
|------|-----|------|
| IORESOURCE_MEM | 0x00000200 | 内存资源 |
| IORESOURCE_IRQ | 0x00000400 | 中断资源 |
| IORESOURCE_DMA | 0x00000800 | DMA通道资源 |
| IORESOURCE_IO | 0x00000100 | I/O端口资源 |
| IORESOURCE_REG | 0x00000300 | 寄存器偏移资源 |
| IORESOURCE_BUS | 0x00001000 | 总线资源 |

---

#### `struct of_device_id` - 设备树匹配表结构体

```c
struct of_device_id {
    char name[32];          /* 节点名称匹配（已废弃） */
    char type[32];          /* 设备类型匹配（已废弃） */
    char compatible[128];   /* compatible字符串匹配 */
    const void *data;       /* 匹配后传递的私有数据 */
};
```

**使用示例**：
```c
static const struct of_device_id xxx_of_match[] = {
    { .compatible = "vendor,xxx-v1", .data = (void *)1 },
    { .compatible = "vendor,xxx-v2", .data = (void *)2 },
    { /* sentinel */ }
};
```

#### `struct device_driver` - 设备驱动核心结构体

```c
struct device_driver {
    const char              *name;              /* 驱动名称 */
    struct bus_type         *bus;               /* 所属总线类型 */
    struct module           *owner;             /* 模块所有者 */
    const char              *mod_name;          /* 模块名称 */
    bool                    suppress_bind_attrs;/* 禁止bind/unbind属性 */
    enum probe_type         probe_type;         /* 探测类型 */
    const struct of_device_id *of_match_table;  /* 设备树匹配表 */
    const struct acpi_device_id *acpi_match_table; /* ACPI匹配表 */
    int (*probe)(struct device *dev);
    int (*remove)(struct device *dev);
    void (*shutdown)(struct device *dev);
    int (*suspend)(struct device *dev, pm_message_t state);
    int (*resume)(struct device *dev);
    const struct attribute_group **groups;
    const struct attribute_group **dev_groups;
    const struct dev_pm_ops *pm;                /* 电源管理操作 */
    void (*coredump)(struct device *dev);
    struct driver_private *p;
};
```

### 3.1 完整的Platform驱动示例（LED驱动）

```c
/*
 * platform_led_driver.c - Platform LED驱动示例
 */

#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/of.h>
#include <linux/of_device.h>
#include <linux/io.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>
#include <linux/gpio/consumer.h>

#define DEVICE_NAME "platform_led"
#define CLASS_NAME  "led_class"

/* 设备私有数据结构 */
struct led_dev {
    struct gpio_desc *led_gpio;     /* LED GPIO描述符 */
    void __iomem *reg_base;         /* 寄存器基地址 */
    int irq;                        /* 中断号 */
    dev_t devno;                    /* 设备号 */
    struct cdev cdev;               /* 字符设备 */
    struct class *class;            /* 设备类 */
    struct device *device;          /* 设备 */
    bool led_state;                 /* LED状态 */
};

/* 文件操作：打开 */
static int led_open(struct inode *inode, struct file *filp)
{
    struct led_dev *led = container_of(inode->i_cdev, struct led_dev, cdev);
    filp->private_data = led;
    pr_info("LED device opened\n");
    return 0;
}

/* 文件操作：关闭 */
static int led_release(struct inode *inode, struct file *filp)
{
    pr_info("LED device closed\n");
    return 0;
}

/* 文件操作：读取 */
static ssize_t led_read(struct file *filp, char __user *buf,
                        size_t count, loff_t *ppos)
{
    struct led_dev *led = filp->private_data;
    char status = led->led_state ? '1' : '0';

    if (*ppos >= 1)
        return 0;

    if (copy_to_user(buf, &status, 1))
        return -EFAULT;

    *ppos += 1;
    return 1;
}

/* 文件操作：写入 */
static ssize_t led_write(struct file *filp, const char __user *buf,
                         size_t count, loff_t *ppos)
{
    struct led_dev *led = filp->private_data;
    char val;

    if (count < 1)
        return -EINVAL;

    if (copy_from_user(&val, buf, 1))
        return -EFAULT;

    if (val == '1') {
        gpiod_set_value(led->led_gpio, 1);
        led->led_state = true;
    } else if (val == '0') {
        gpiod_set_value(led->led_gpio, 0);
        led->led_state = false;
    }

    return count;
}

/* 文件操作集 */
static const struct file_operations led_fops = {
    .owner   = THIS_MODULE,
    .open    = led_open,
    .release = led_release,
    .read    = led_read,
    .write   = led_write,
};

/* 设备树匹配表 */
static const struct of_device_id led_of_match[] = {
    { .compatible = "vendor,platform-led", .data = (void *)0 },
    { .compatible = "vendor,platform-led-v2", .data = (void *)1 },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(of, led_of_match);

/* 平台设备ID表（传统匹配方式） */
static const struct platform_device_id led_id_table[] = {
    { "platform-led", 0 },
    { "platform-led-v2", 1 },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(platform, led_id_table);

/* probe函数 - 设备匹配成功后调用 */
static int led_probe(struct platform_device *pdev)
{
    struct led_dev *led;
    struct resource *res;
    const struct of_device_id *match;
    int ret;

    dev_info(&pdev->dev, "LED probe called\n");

    /* 1. 分配设备私有数据 */
    led = devm_kzalloc(&pdev->dev, sizeof(*led), GFP_KERNEL);
    if (!led)
        return -ENOMEM;

    /* 2. 获取设备树匹配数据 */
    match = of_match_device(led_of_match, &pdev->dev);
    if (match) {
        dev_info(&pdev->dev, "Matched with data: %ld\n", 
                 (long)match->data);
    }

    /* 3. 获取并映射内存资源（可选） */
    res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    if (res) {
        led->reg_base = devm_ioremap_resource(&pdev->dev, res);
        if (IS_ERR(led->reg_base)) {
            dev_warn(&pdev->dev, "Failed to map registers\n");
            /* 非致命错误，继续执行 */
        } else {
            dev_info(&pdev->dev, "Mapped registers at %pR\n", res);
        }
    }

    /* 4. 获取中断资源（可选） */
    led->irq = platform_get_irq_optional(pdev, 0);
    if (led->irq >= 0) {
        dev_info(&pdev->dev, "Got IRQ: %d\n", led->irq);
    }

    /* 5. 获取GPIO资源 */
    led->led_gpio = devm_gpiod_get(&pdev->dev, "led", GPIOD_OUT_LOW);
    if (IS_ERR(led->led_gpio)) {
        dev_err(&pdev->dev, "Failed to get LED GPIO\n");
        return PTR_ERR(led->led_gpio);
    }

    /* 6. 分配设备号 */
    ret = alloc_chrdev_region(&led->devno, 0, 1, DEVICE_NAME);
    if (ret < 0) {
        dev_err(&pdev->dev, "Failed to allocate device number\n");
        return ret;
    }

    /* 7. 初始化并添加字符设备 */
    cdev_init(&led->cdev, &led_fops);
    led->cdev.owner = THIS_MODULE;
    ret = cdev_add(&led->cdev, led->devno, 1);
    if (ret) {
        dev_err(&pdev->dev, "Failed to add cdev\n");
        goto err_cdev;
    }

    /* 8. 创建设备类 */
    led->class = class_create(THIS_MODULE, CLASS_NAME);
    if (IS_ERR(led->class)) {
        ret = PTR_ERR(led->class);
        dev_err(&pdev->dev, "Failed to create class\n");
        goto err_class;
    }

    /* 9. 创建设备节点 */
    led->device = device_create(led->class, &pdev->dev, led->devno,
                                NULL, DEVICE_NAME);
    if (IS_ERR(led->device)) {
        ret = PTR_ERR(led->device);
        dev_err(&pdev->dev, "Failed to create device\n");
        goto err_device;
    }

    /* 10. 保存私有数据 */
    platform_set_drvdata(pdev, led);

    dev_info(&pdev->dev, "LED driver probed successfully\n");
    return 0;

err_device:
    class_destroy(led->class);
err_class:
    cdev_del(&led->cdev);
err_cdev:
    unregister_chrdev_region(led->devno, 1);
    return ret;
}

/* remove函数 - 设备移除时调用 */
static int led_remove(struct platform_device *pdev)
{
    struct led_dev *led = platform_get_drvdata(pdev);

    /* 关闭LED */
    gpiod_set_value(led->led_gpio, 0);

    /* 按照初始化的逆序清理资源 */
    device_destroy(led->class, led->devno);
    class_destroy(led->class);
    cdev_del(&led->cdev);
    unregister_chrdev_region(led->devno, 1);

    dev_info(&pdev->dev, "LED driver removed\n");
    return 0;
}

/* 电源管理 - 挂起 */
static int led_suspend(struct device *dev)
{
    struct led_dev *led = dev_get_drvdata(dev);
    
    /* 挂起时关闭LED */
    gpiod_set_value(led->led_gpio, 0);
    dev_info(dev, "LED suspended\n");
    return 0;
}

/* 电源管理 - 恢复 */
static int led_resume(struct device *dev)
{
    struct led_dev *led = dev_get_drvdata(dev);
    
    /* 恢复LED状态 */
    gpiod_set_value(led->led_gpio, led->led_state);
    dev_info(dev, "LED resumed\n");
    return 0;
}

/* 电源管理操作集 */
static SIMPLE_DEV_PM_OPS(led_pm_ops, led_suspend, led_resume);

/* Platform驱动结构体 */
static struct platform_driver led_driver = {
    .probe  = led_probe,
    .remove = led_remove,
    .driver = {
        .name = "platform-led",
        .of_match_table = led_of_match,
        .pm = &led_pm_ops,
    },
    .id_table = led_id_table,
};

/* 使用简化宏注册驱动 */
module_platform_driver(led_driver);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("Platform LED Driver Example");
MODULE_VERSION("1.0");
```
